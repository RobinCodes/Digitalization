A bőr a testsúlyunk 15-20\%-át teszi ki.
\subsection{Többrétegű laphám}
\begin{itemize}
    \item Alul folyamatosan osztódó sejtek
\end{itemize}

\subsection{A bőr 3 rétege}
\begin{itemize}
    \item Hám \begin{itemize}
        \item Nincsenek erek, táplálása az irhából
        \item Szaruréteg
        \item Pigmentsejtek - melanint termelnek, amely véd az UV sugárzástól (melanin, bőrszín) 
    \end{itemize}
    \item Irha \begin{itemize}
        \item Hajszál gyökerei (szőrtüsző) 
        \item Verejtékmirigy
        \item Rengeteg idegvégződés
        \item Erek
    \end{itemize}
    \item Bőralja
    \begin{itemize}
        \item Zsírszövetben gazdag
        \item Összekapcsolja a bőrt az alatta lévő izomcsoportokkal
    \end{itemize}
\end{itemize}

\subsection{Receptorok a bőrben}
\begin{itemize}
    \item Elosztásuk nem egyenletes a testben
    \item Az ingerület agytörzsbe jut
    \item Szabad idegvégződés - Fájdalom - felhívja a figyelmet a veszélyre, és ösztönöz annak elkerülésére (az idegvégződés roncsolódása érzékel)
    \item Tapintó ideg - finom nyomás - ujjak hegyén találhatóak nagy mennyiségben - gyors nyomásváltozásra nagyon érzékenyek
\end{itemize}
