\section{Graphs}
\begin{definition}[Graphs]
A graph is made up of vertices and edges. An edge either connects two vertices or one vertex to itself.
\end{definition}

\begin{definition}[Loop]
A loop is an edge which connects a vertex to itself.
\end{definition}
